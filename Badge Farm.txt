local function fr(path)
    for i = 1, 5 do
        local ok, c = pcall(readfile, path)
        if ok and c then
            return c
        end
        task.wait(0.3)
    end
    return nil
end

local function fw(path, content)
    for i = 1, 3 do
        local ok = pcall(writefile, path, content)
        if ok then
            return true
        end
        task.wait(0.2)
    end
    return false
end

local function fe(path)
    local ok, ex = pcall(isfile, path)
    return ok and ex
end

local function fd(path)
    pcall(delfile, path)
end

local function listLoad(path)
    local set = {}
    local content = fr(path)
    if content then
        for line in content:gmatch("[^\r\n]+") do
            local n = tonumber(line)
            if n then
                set[n] = true
            end
        end
    end
    return set
end

local function listAdd(path, n)
    local set = listLoad(path)
    if not set[n] then
        set[n] = true
        local lines = {}
        for k in pairs(set) do
            table.insert(lines, tostring(k))
        end
        fw(path, table.concat(lines, "\n"))
    end
end

local function listCount(path)
    local set = listLoad(path)
    local c = 0
    for _ in pairs(set) do
        c = c + 1
    end
    return c
end

-- ════ HTTP ════

local function httpGet(url)
    local result = nil
    pcall(function()
        if type(request) == "function" then
            local res = request({Url = url, Method = "GET"})
            if res and res.StatusCode == 200 then
                result = res.Body
            end
        end
    end)
    if not result then
        pcall(function()
            result = game:HttpGet(url)
        end)
    end
    return result
end

local function jsonDecode(str)
    if not str or str == "" then
        return nil
    end
    local ok, data = pcall(function()
        return game:GetService("HttpService"):JSONDecode(str)
    end)
    if ok then
        return data
    end
    return nil
end

local function getUniverseId(placeId)
    local body = httpGet("https://apis.roblox.com/universes/v1/places/" .. placeId .. "/universe")
    if body then
        local data = jsonDecode(body)
        if data and data.universeId then
            return data.universeId
        end
    end
    return nil
end

-- ════ BADGES ════

local BadgeService = game:GetService("BadgeService")

local function getUniverseBadges(universeId)
    local badges = {}
    local body = httpGet("https://badges.roblox.com/v1/universes/" .. universeId .. "/badges?limit=100&sortOrder=Asc")
    if body then
        local data = jsonDecode(body)
        if data and data.data then
            for _, badge in ipairs(data.data) do
                table.insert(badges, {
                    id = badge.id,
                    name = badge.name or "unknown",
                    enabled = badge.enabled ~= false,
                })
            end
        end
    end
    return badges
end

local function userHasBadge(userId, badgeId)
    local ok, has = pcall(function()
        return BadgeService:UserHasBadgeAsync(userId, badgeId)
    end)
    if ok then
        return has
    end
    return nil
end

local function isWelcomeBadge(badgeName)
    local n = badgeName:lower()
    if n:find("welcome") then return true end
    if n:find("you played") then return true end
    if n:find("you joined") then return true end
    if n:find("visited") then return true end
    if n:find("joined") then return true end
    if n:find("first join") then return true end
    return false
end

local function findWelcomeBadge(placeId)
    local uid = getUniverseId(placeId)
    if not uid then
        return nil, nil
    end
    local badges = getUniverseBadges(uid)
    for _, b in ipairs(badges) do
        if b.enabled and isWelcomeBadge(b.name) then
            return b, uid
        end
    end
    return nil, uid
end

-- ════ TELEPORT ════

local spoofGameId = nil
local hookInstalled = false

local function installHook()
    if hookInstalled then
        return
    end
    local oldIdx
    local newFunc
    newFunc = function(t, k)
        if k == "GameId" and spoofGameId then
            return spoofGameId
        end
        if oldIdx then
            return oldIdx(t, k)
        end
    end
    if type(newcclosure) == "function" then
        pcall(function()
            newFunc = newcclosure(newFunc)
        end)
    end
    local ok, result = pcall(hookmetamethod, game, "__index", newFunc)
    if ok then
        oldIdx = result
        hookInstalled = true
        log("[hook] installed")
    else
        log("[hook] failed")
    end
end

local function tryTeleport(placeId)
    local TS = game:GetService("TeleportService")
    local lp = Players.LocalPlayer
    if not lp then
        return false
    end

    local uid = getUniverseId(placeId)
    log("target: " .. placeId .. " (universe " .. (uid or "?") .. ")")

    pcall(setfflag, "FFlagTeleportServiceRestrictCrossUniverse", "False")
    pcall(setfflag, "FFlagTeleportCrossUniverseCheck", "False")
    pcall(setfflag, "FFlagTeleportServiceCrossUniverse", "False")
    pcall(setfflag, "FFlagRestrictCrossUniverseTeleport", "False")

    installHook()

    local before = game.PlaceId
    log("currently in: " .. before)

    -- method 1: Teleport
    if hookInstalled and uid then
        spoofGameId = uid
    end

    log("[1/2] Teleport...")
    local ok1, err1 = pcall(function()
        TS:Teleport(placeId)
    end)

    if ok1 then
        log("[1/2] call ok, waiting " .. TP_WAIT .. "s...")
        spoofGameId = nil
        local s = tick()
        while tick() - s < TP_WAIT do
            task.wait(0.5)
            if game.PlaceId ~= before then
                log("[1/2] PLACE CHANGED -- SUCCESS")
                return true
            end
        end
        log("[1/2] failed (still in " .. game.PlaceId .. ")")
    else
        log("[1/2] errored: " .. tostring(err1))
        spoofGameId = nil
    end

    -- method 2: PlaceInstance
    log("[2/2] PlaceInstance...")
    local sb = httpGet("https://games.roblox.com/v1/games/" .. placeId .. "/servers/Public?sortOrder=Asc&limit=100")
    if sb then
        local sd = jsonDecode(sb)
        if sd and sd.data and sd.data[1] then
            local jobId = sd.data[1].id
            log("[2/2] server: " .. tostring(jobId))

            if hookInstalled and uid then
                spoofGameId = uid
            end

            local ok2, err2 = pcall(function()
                TS:TeleportToPlaceInstance(placeId, jobId, lp)
            end)

            if ok2 then
                log("[2/2] call ok, waiting " .. TP_WAIT .. "s...")
                spoofGameId = nil
                local s = tick()
                while tick() - s < TP_WAIT do
                    task.wait(0.5)
                    if game.PlaceId ~= before then
                        log("[2/2] PLACE CHANGED -- SUCCESS")
                        return true
                    end
                end
                log("[2/2] failed")
            else
                log("[2/2] errored: " .. tostring(err2))
                spoofGameId = nil
            end
        else
            log("[2/2] no servers found")
        end
    else
        log("[2/2] could not fetch servers")
    end

    spoofGameId = nil
    log("TELEPORT FAILED for " .. placeId)
    return false
end

-- ════ FIND NEXT ════

local function findNextGame()
    local visited = listLoad(F_VISITED)
    local failed = listLoad(F_FAILED)
    local skip = listLoad(F_SKIP)
    local normal = {}
    local lowPriority = {}
    for i, pid in ipairs(IDS) do
        if not visited[pid] and not skip[pid] then
            if PERMA_LOW[pid] or failed[pid] then
                table.insert(lowPriority, {index = i, id = pid})
            else
                table.insert(normal, {index = i, id = pid})
            end
        end
    end
    if #normal > 0 then
        return normal[1].index, normal[1].id
    end
    if #lowPriority > 0 then
        return lowPriority[1].index, lowPriority[1].id
    end
    return nil, nil
end

-- ════ PROCESS TARGET GAME ════

local function processTargetGame(placeId)
    -- shared lock: whichever script (main or engine) runs first claims the game
    if _G.BF_PROCESSED == placeId then
        log("already processed " .. placeId .. " this session (lock)")
        return
    end
    _G.BF_PROCESSED = placeId

    log("TARGET GAME: " .. placeId)

    local welcome, uid = findWelcomeBadge(placeId)
    if welcome then
        local has = userHasBadge(PLAYER_ID, welcome.id)
        if has == true then
            log('welcome already owned: "' .. welcome.name .. '"')
        else
            log('waiting ' .. BADGE_WAIT .. 's for "' .. welcome.name .. '"...')
            local start = tick()
            local awarded = false
            while tick() - start < BADGE_WAIT do
                local h = userHasBadge(PLAYER_ID, welcome.id)
                if h == true then
                    log('WELCOME BADGE AWARDED: "' .. welcome.name .. '"')
                    awarded = true
                    break
                end
                task.wait(2)
            end
            if not awarded then
                log("not awarded in " .. BADGE_WAIT .. "s, marking visited anyway")
            end
        end
    else
        log("no welcome badge, waiting " .. BADGE_WAIT .. "s")
        task.wait(BADGE_WAIT)
    end

    listAdd(F_VISITED, placeId)
    log("VISITED: " .. placeId)

    local vCount = listCount(F_VISITED)
    log("progress: " .. vCount .. "/" .. MIN_VISITS)

    if vCount >= MIN_VISITS then
        log("GOAL REACHED for " .. PLAYER_NAME)
        fd(F_STATE)
        task.wait(1)
        local lp = Players.LocalPlayer
        if lp then
            lp:Kick("BADGE OBTAINING COMPLETE -- " .. vCount .. "/" .. MIN_VISITS)
        end
    end
end

local function main()
    -- prevent this script from running twice in same session
    if _G.BF_MAIN_RAN then
        log("main already ran this session, skipping")
        return
    end
    _G.BF_MAIN_RAN = true

    log("badge farm V" .. VERSION)
    log("account: " .. PLAYER_NAME .. " (" .. PLAYER_ID .. ")")

    if not fe(F_START) then
        fw(F_START, tostring(os.time()))
    end

    local lp = Players.LocalPlayer
    local state = fr(F_STATE)
    state = state and tonumber(state)

    -- save engine + queue
    saveEngineAndQueue()

    -- ══ FIRST RUN ══
    if not state then
        log("FIRST RUN for " .. PLAYER_NAME)

        while not game:IsLoaded() do
            task.wait(0.5)
        end
        task.wait(2)

        fw(F_STATE, "1")

        -- preflight: check welcome badges for all games
        log("--- preflight ---")
        for i, placeId in ipairs(IDS) do
            local welcome, uid = findWelcomeBadge(placeId)
            if welcome then
                local has = userHasBadge(PLAYER_ID, welcome.id)
                if has == true then
                    log('  ' .. placeId .. ': welcome owned ("' .. welcome.name .. '") -- skip')
                    listAdd(F_SKIP, placeId)
                    listAdd(F_VISITED, placeId)
                else
                    log('  ' .. placeId .. ': welcome needed ("' .. welcome.name .. '")')
                end
            else
                log('  ' .. placeId .. ': no welcome badge')
            end
            task.wait(0.3)
        end
        log("--- preflight done ---")
        log("")

        local visitedCount = listCount(F_VISITED)
        if visitedCount >= MIN_VISITS then
            log("goal already met (" .. visitedCount .. "/" .. MIN_VISITS .. ")")
            fd(F_STATE)
            task.wait(1)
            if lp then
                lp:Kick("BADGE OBTAINING COMPLETE -- " .. visitedCount .. "/" .. MIN_VISITS)
            end
            return
        end

        local nextIdx, nextId = findNextGame()
        if not nextId then
            log("all games have welcome badges already")
            fd(F_STATE)
            task.wait(1)
            if lp then
                lp:Kick("BADGE OBTAINING COMPLETE")
            end
            return
        end

        log("starting game " .. nextIdx .. "/" .. TOTAL .. ": " .. nextId)
        local teleported = tryTeleport(nextId)

        if not teleported then
            log("BLOCKED: " .. nextId)
            listAdd(F_FAILED, nextId)

            local n2Idx, n2Id = findNextGame()
            if n2Id then
                log("trying: " .. n2Id)
                local tp2 = tryTeleport(n2Id)
                if not tp2 then
                    listAdd(F_FAILED, n2Id)
                    log("multiple blocks, re-execute to retry")
                end
            end
        end
        flushLog()
        return
    end

    -- ══ RESUME ══
    log("RESUMING for " .. PLAYER_NAME)

    while not game:IsLoaded() do
        task.wait(0.5)
    end

    if lp and not lp.Character then
        local spawned = false
        local c = lp.CharacterAdded:Connect(function()
            spawned = true
        end)
        local s = tick()
        while not spawned and tick() - s < 30 do
            task.wait(0.5)
        end
        c:Disconnect()
    end
    task.wait(2)

    local currentPlace = game.PlaceId
    log("current game: " .. currentPlace)

    local inList = false
    for _, pid in ipairs(IDS) do
        if pid == currentPlace then
            inList = true
            break
        end
    end

    if inList then
        local visitedSet = listLoad(F_VISITED)
        if not visitedSet[currentPlace] then
            processTargetGame(currentPlace)
        else
            log("already visited " .. currentPlace)
        end
    else
        log("not a target game")
    end

    local visitedCount = listCount(F_VISITED)
    local skipCount = listCount(F_SKIP)

    if visitedCount + skipCount >= TOTAL then
        log("all games done")
        fd(F_STATE)
        task.wait(1)
        if lp then
            lp:Kick("BADGE OBTAINING COMPLETE -- " .. visitedCount .. " visits")
        end
        return
    end

    local nextIdx, nextId = findNextGame()
    if not nextId then
        log("no more games")
        fd(F_STATE)
        return
    end

    log("next: " .. nextId)
    local okTp = tryTeleport(nextId)

    if not okTp then
        log("BLOCKED: " .. nextId)
        listAdd(F_FAILED, nextId)

        local n2Idx, n2Id = findNextGame()
        if n2Id then
            log("trying: " .. n2Id)
            local ok2 = tryTeleport(n2Id)
            if not ok2 then
                listAdd(F_FAILED, n2Id)
                log("multiple blocks, re-execute to retry")
            end
        else
            log("all blocked, re-execute or join manually")
        end
    end

    flushLog()
end

-- ════ RUN ════

local ok, err = pcall(main)
if not ok then
    log("FATAL: " .. tostring(err))
end

log("done")
flushLog()
